58: -- Record only writes that actually happened. BEFORE INSERT also runs for
59: -- ON CONFLICT DO NOTHING; recording there would invalidate exact inbox replays.
60: CREATE FUNCTION public.research_source_fence_record_v2() RETURNS trigger
61:  LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog,public AS $$
62: DECLARE old_doc jsonb; new_doc jsonb; eid bigint; roots text[];
63: BEGIN
64:  IF current_setting('transaction_isolation')<>'read committed' THEN RAISE EXCEPTION 'source_fence_read_committed_required'; END IF;
65:  PERFORM pg_advisory_xact_lock(610091002::bigint);
66:  IF TG_OP<>'INSERT' THEN old_doc:=to_jsonb(OLD); END IF;
67:  IF TG_OP<>'DELETE' THEN new_doc:=to_jsonb(NEW); END IF;
68:  IF old_doc IS NOT DISTINCT FROM new_doc THEN RETURN NEW; END IF;
69:  roots:=ARRAY[public.research_source_root_v2(old_doc),public.research_source_root_v2(new_doc)];
70:  INSERT INTO public.research_source_fence_events_v2(document_id,operation,before_hash,after_hash)
71:  VALUES(COALESCE(NEW.id,OLD.id),TG_OP,
72:   CASE WHEN old_doc IS NOT NULL THEN encode(sha256(convert_to(old_doc::text,'UTF8')),'hex') END,
73:   CASE WHEN new_doc IS NOT NULL THEN encode(sha256(convert_to(new_doc::text,'UTF8')),'hex') END) RETURNING event_id INTO eid;
74:  INSERT INTO public.research_source_seal_invalidations_v2(seal_id,event_id)
75:  SELECT s.seal_id,eid FROM public.research_source_seals_v2 s
76:  WHERE EXISTS(SELECT FROM jsonb_array_elements(s.documents) d WHERE d->>'root'=ANY(roots));
77:  IF TG_OP='DELETE' THEN RETURN OLD; END IF;
78:  RETURN NEW;
79: END $$;
80: ALTER FUNCTION public.research_source_fence_record_v2() OWNER TO research_source_fence_owner_v2;
81: REVOKE ALL ON FUNCTION public.research_source_fence_record_v2() FROM PUBLIC,anon,authenticated,service_role;
82: CREATE TRIGGER research_source_fence_record_v2 AFTER INSERT OR UPDATE OR DELETE ON public.source_raw_documents
83:  FOR EACH ROW EXECUTE FUNCTION public.research_source_fence_record_v2();
