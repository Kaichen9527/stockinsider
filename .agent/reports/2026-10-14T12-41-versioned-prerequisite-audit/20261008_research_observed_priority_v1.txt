252: CREATE OR REPLACE FUNCTION public.claim_research_deep_job_v1(p_owner TEXT)
253: RETURNS TABLE(job_id UUID, symbol TEXT, priority_run_id UUID, attempt INTEGER, lease_expires_at TIMESTAMPTZ)
254: LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp
255: AS $function$
256: DECLARE v_job public.research_deep_jobs_v1; v_reservation uuid;
257: BEGIN
258:   IF p_owner IS NULL OR length(trim(p_owner)) < 3 OR length(p_owner) > 120 THEN
259:     RAISE EXCEPTION 'research_deep_owner_invalid';
260:   END IF;
261:   PERFORM pg_advisory_xact_lock(2409, 6002);
262:   PERFORM public.reap_expired_research_deep_jobs_v2();
263:   IF EXISTS (SELECT 1 FROM public.research_deep_jobs_v1 WHERE status='running')
264:     OR (SELECT count(*) FROM public.research_deep_job_attempts_v1
265:        WHERE (claimed_at AT TIME ZONE 'Asia/Taipei')::date =
266:          (clock_timestamp() AT TIME ZONE 'Asia/Taipei')::date) >= 4 THEN
267:     RETURN;
268:   END IF;
269:   SELECT * INTO v_job FROM public.research_deep_jobs_v1 AS jobs
270:     WHERE jobs.research_scope='formal_v1' AND jobs.status='queued' AND jobs.attempts < 3
271:     ORDER BY jobs.week_start, jobs.queue_rank, jobs.created_at, jobs.job_id
272:     FOR UPDATE SKIP LOCKED LIMIT 1;
273:   IF NOT FOUND THEN RETURN; END IF;
274:   SELECT reservation_id INTO v_reservation FROM public.reserve_research_model_v1(
275:     'company_research',p_owner,'deep:'||v_job.job_id||':'||(v_job.attempts+1)::text);
276:   IF v_reservation IS NULL THEN RETURN; END IF;
277:   UPDATE public.research_deep_jobs_v1 SET status='running', attempts=attempts+1,
278:     lease_owner=p_owner, lease_expires_at=clock_timestamp()+interval '30 minutes'
279:     WHERE research_deep_jobs_v1.job_id=v_job.job_id
280:     RETURNING * INTO v_job;
281:   INSERT INTO public.research_deep_job_attempts_v1(job_id,attempt,owner,lease_expires_at)
282:     VALUES (v_job.job_id,v_job.attempts,p_owner,v_job.lease_expires_at);
283:   RETURN QUERY SELECT v_job.job_id,v_job.symbol,v_job.priority_run_id,v_job.attempts,v_job.lease_expires_at;
284: END $function$;
