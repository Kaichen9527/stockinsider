178: CREATE OR REPLACE FUNCTION public.finish_research_deep_job_v2(
179:   p_job_id UUID, p_owner TEXT, p_attempt INTEGER,
180:   p_success BOOLEAN, p_receipt_id UUID, p_reason TEXT
181: )
182: RETURNS BOOLEAN
183: LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp
184: AS $function$
185: DECLARE v_job public.research_deep_jobs_v1;
186: BEGIN
187:   PERFORM pg_advisory_xact_lock(2409, 6002);
188:   SELECT * INTO v_job FROM public.research_deep_jobs_v1 WHERE job_id=p_job_id AND research_scope='formal_v1' FOR UPDATE;
189:   IF FOUND AND p_success AND v_job.status='completed' AND v_job.receipt_id=p_receipt_id
190:     AND v_job.attempts=p_attempt AND v_job.completion_owner=p_owner THEN
191:     RETURN TRUE;
192:   END IF;
193:   IF NOT FOUND OR v_job.status <> 'running' OR v_job.lease_owner IS DISTINCT FROM p_owner
194:     OR v_job.attempts<>p_attempt OR v_job.lease_expires_at <= clock_timestamp() THEN
195:     RAISE EXCEPTION 'research_deep_job_lease_lost';
196:   END IF;
197:   IF p_success THEN RAISE EXCEPTION 'research_deep_publication_receipt_required'; END IF;
198:   -- Release the publication lease with the same deep attempt; a reported
199:   -- failure must not strand the next attempt behind an active old lease.
200:   UPDATE public.candidate_dossier_outbox_v5 SET
201:     status=CASE WHEN v_job.attempts<3 THEN 'queued' ELSE 'failed' END,
202:     lease_owner=NULL, lease_expires_at=NULL, next_attempt_at=NULL,
203:     last_error=left(coalesce(p_reason,'research_failed'),500), updated_at=clock_timestamp()
204:     WHERE publication_kind='deep' AND deep_job_id=p_job_id AND deep_attempt=p_attempt
205:       AND status='running';
206:   UPDATE public.research_deep_jobs_v1 SET
207:     status=CASE WHEN attempts < 3 THEN 'queued' ELSE 'failed' END,
208:     lease_owner=NULL,lease_expires_at=NULL,
209:     receipt_id=NULL,
210:     terminal_reason=left(coalesce(p_reason,'research_failed'),500),
211:     finished_at=CASE WHEN attempts >= 3 THEN clock_timestamp() ELSE NULL END
212:   WHERE research_deep_jobs_v1.job_id=p_job_id;
213:   PERFORM public.finish_research_model_v1(r.reservation_id,p_owner,
214:     CASE WHEN p_success THEN 'completed' ELSE 'failed' END,
215:     encode(sha256(convert_to(COALESCE(p_receipt_id::text,p_reason),'UTF8')),'hex'))
216:   FROM public.research_model_reservations_v1 r WHERE r.role='company_research'
217:     AND r.work_key='deep:'||p_job_id||':'||p_attempt::text
218:     AND NOT EXISTS(SELECT 1 FROM public.research_model_completions_v1 c WHERE c.reservation_id=r.reservation_id);
219:   RETURN TRUE;
220: END $function$;

259:   IF NOT FOUND OR v_job.status<>'running' OR v_job.lease_owner IS DISTINCT FROM p_deep_owner
260:     OR v_job.attempts<>p_deep_attempt OR v_job.lease_expires_at<=clock_timestamp() THEN
261:     RAISE EXCEPTION 'research_deep_job_lease_lost';
262:   END IF;
263:   SELECT * INTO v_outbox FROM public.candidate_dossier_outbox_v5 outbox
264:     WHERE outbox.job_id=p_job_id FOR UPDATE;
265:   IF NOT FOUND OR v_outbox.publication_kind<>'deep' OR v_outbox.status<>'running'
266:     OR v_outbox.lease_expires_at<=clock_timestamp() OR v_outbox.deep_job_id IS DISTINCT FROM p_deep_job_id
267:     OR v_outbox.deep_attempt IS DISTINCT FROM p_deep_attempt
268:     OR v_outbox.lease_owner IS DISTINCT FROM p_owner
269:     OR v_outbox.bundle_id IS DISTINCT FROM p_bundle_id
270:     OR v_outbox.revision_id IS DISTINCT FROM p_revision_id
271:     OR v_outbox.input_hash IS DISTINCT FROM p_input_hash THEN
272:     RAISE EXCEPTION 'research_deep_outbox_binding_missing';
273:   END IF;
274:   SELECT detail.stock_id INTO v_stock FROM public.candidate_detail_snapshots detail
275:     WHERE detail.id=p_revision_id;
276:   IF v_stock IS DISTINCT FROM v_job.stock_id THEN
277:     RAISE EXCEPTION 'research_deep_stock_revision_mismatch';
278:   END IF;
279:   IF p_validation_status='valid' AND (
280:     p_content->'deepResearch'->>'articleHash' IS DISTINCT FROM p_article_hash
281:     OR NOT EXISTS (
282:       SELECT 1 FROM public.candidate_deep_article_reviews_v1 review
283:       WHERE review.id=p_deep_review_id AND review.revision_id=p_revision_id
284:         AND review.input_hash=p_input_hash AND review.article_hash=p_article_hash
285:         AND review.decision='accepted' AND review.reviewed_at>=v_job.created_at
286:         AND EXISTS(SELECT 1 FROM public.research_model_completions_v1 c
287:           JOIN public.research_model_reservations_v1 r USING(reservation_id)
288:           WHERE c.reservation_id=review.model_reservation_id AND c.outcome='completed'
289:             AND c.result_hash=p_article_hash AND r.role='counter_review'
290:             AND r.work_key='deep-review:'||p_deep_job_id||':'||p_deep_attempt||':'||p_article_hash)
291:     )
292:   ) THEN
293:     RAISE EXCEPTION 'research_deep_exact_review_missing';
294:   END IF;
295:   SELECT * INTO v_receipt FROM public.record_candidate_dossier_submission_v4(
296:     p_bundle_id,p_revision_id,p_input_hash,p_submission_hash,p_content,
297:     p_claims,p_source_references,p_claim_fact_map,p_validation_status,p_rejection_reasons
298:   );
299:   IF NOT FOUND THEN RAISE EXCEPTION 'research_deep_publication_receipt_missing'; END IF;
300:   UPDATE public.candidate_dossier_outbox_v5 SET
301:     status=v_receipt.status, lease_owner=NULL, lease_expires_at=NULL,
302:     receipt_id=CASE WHEN v_receipt.status='accepted' THEN v_receipt.submission_id ELSE NULL END,
303:     last_submission_hash=p_submission_hash,
304:     last_error=CASE WHEN v_receipt.status='rejected' THEN v_receipt.rejection_reasons::text ELSE NULL END,
305:     updated_at=clock_timestamp() WHERE job_id=p_job_id;
306:   UPDATE public.research_deep_jobs_v1 SET
307:     status=CASE WHEN v_receipt.status='accepted' THEN 'completed'
308:       WHEN attempts<3 THEN 'queued' ELSE 'failed' END,
309:     lease_owner=NULL,lease_expires_at=NULL,
310:     receipt_id=CASE WHEN v_receipt.status='accepted' THEN v_receipt.submission_id ELSE NULL END,
311:     completion_owner=CASE WHEN v_receipt.status='accepted' THEN p_deep_owner ELSE NULL END,
312:     completion_outbox_owner=CASE WHEN v_receipt.status='accepted' THEN p_owner ELSE NULL END,
313:     completion_outbox_job_id=CASE WHEN v_receipt.status='accepted' THEN p_job_id ELSE NULL END,
314:     completion_review_id=CASE WHEN v_receipt.status='accepted' THEN p_deep_review_id ELSE NULL END,
315:     completion_article_hash=CASE WHEN v_receipt.status='accepted' THEN p_article_hash ELSE NULL END,
316:     completion_submission_hash=CASE WHEN v_receipt.status='accepted' THEN p_submission_hash ELSE NULL END,
317:     terminal_reason=CASE WHEN v_receipt.status='accepted' THEN NULL ELSE 'dossier_rejected' END,
318:     finished_at=CASE WHEN v_receipt.status='accepted' OR attempts>=3 THEN clock_timestamp() ELSE NULL END
319:     WHERE research_deep_jobs_v1.job_id=p_deep_job_id;
320:   PERFORM public.finish_research_model_v1(r.reservation_id,p_deep_owner,
321:     CASE WHEN v_receipt.status='accepted' THEN 'completed' ELSE 'failed' END,p_submission_hash)
322:   FROM public.research_model_reservations_v1 r WHERE r.role='company_research'
323:     AND r.work_key='deep:'||p_deep_job_id||':'||p_deep_attempt::text
324:     AND NOT EXISTS(SELECT 1 FROM public.research_model_completions_v1 c WHERE c.reservation_id=r.reservation_id);
325:   RETURN QUERY SELECT v_receipt.submission_id,v_receipt.dossier_id,v_receipt.status,
326:     v_receipt.rejection_reasons,v_receipt.idempotent_replay;
