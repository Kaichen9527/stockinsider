import assert from 'node:assert/strict';
import fs from 'node:fs';
import {evaluateLaunchOutcome,LAUNCH_SIMULATION_COSTS} from './web/src/lib/vnext/launch-outcomes.ts';

const hash=(x)=>x.repeat(64);
const iso=(n)=>new Date(Date.UTC(2025,0,n)).toISOString().slice(0,10);
const signalRun='11111111-1111-4111-8111-111111111111';
const fixture=(count=120,changes={})=>{
  const sessions=Array.from({length:count},(_,i)=>{
    const session=iso(i+2);return {session,openAt:session+'T09:00:00+08:00',closeAt:session+'T13:30:00+08:00',
      knownAt:session+'T15:00:00+08:00',status:'completed'};
  });
  const bars=sessions.map(({session,knownAt})=>({session,availableAt:knownAt,priceRevision:hash('b'),basisRevision:hash('c'),
    rawOpen:'100',open:'100',high:'100',low:'100',close:'100'}));
  const input={scope:'published_rank',signal:{rankingRunId:signalRun,firstRankingRunId:signalRun,symbol:'1234',market:'TPEX',
   session:iso(1),cutoff:iso(1)+'T21:30:00+08:00',inputHash:hash('d'),priceRevision:hash('a'),rawClose:'100',normalizedClose:'100'},
   asOf:(sessions.at(-1)?.session??iso(1))+'T21:45:00+08:00',calendarRevision:hash('e'),basisRevision:hash('c'),
   sessions,bars,opening:sessions[0]?{session:sessions[0].session,knownAt:sessions[0].knownAt,revision:hash('f'),status:'modeled_fill',rawOpen:'100',priceRevision:hash('b'),reason:null}:null};
  return {...input,...changes};
};
const record=[];
function test(name,fn){fn();record.push(name);}
test('single identity across replay and evaluation cutoff',()=>{
 const input=fixture();
 const a=evaluateLaunchOutcome(input),b=evaluateLaunchOutcome(structuredClone(input));
 assert.deepEqual(a,b);
 assert.equal(a.research.barrier20.state,'not_hit');
 assert.deepEqual(a.research.horizons.map(h=>[h.horizon,h.state,h.elapsedSessions]),[[20,'ready',20],[60,'ready',60],[120,'ready',120]]);
 assert.equal(a.research.horizons[2].grossPriceReturnPct,0);
 assert(a.execution.horizons[2].netSimulatedReturnPct<0);
 assert.equal(a.costs.sellTaxPpm,3000);
 assert.notEqual(evaluateLaunchOutcome({...input,scope:'historical_price_replay'}).key,a.key);
 const repeated={...input,signal:{...input.signal,firstRankingRunId:'22222222-2222-4222-8222-222222222222'}};
 assert.equal(evaluateLaunchOutcome(repeated).repeatSignal,true);
});
test('exact threshold inclusivity and distinct net simulator',()=>{
 const i=fixture();Object.assign(i.bars[19],{high:'120',close:'120'});
 const x=evaluateLaunchOutcome(i);
 assert.equal(x.research.barrier20.state,'up_first');
 assert.equal(x.research.barrier20.position,20);
 assert.equal(x.execution.grossBarrier20.state,'up_first');
 assert.equal(x.execution.netBarrier20.state,'not_hit');
 assert.equal(x.research.horizons[0].grossPriceReturnPct,20);
 assert(x.execution.horizons[0].netSimulatedReturnPct<20);
});
test('down first holds even after much higher recovery',()=>{
 const i=fixture();i.bars[2].low='85';i.bars[3].high='130';
 const x=evaluateLaunchOutcome(i);assert.equal(x.research.barrier20.state,'down_first');assert.equal(x.research.barrier20.position,3);
});
test('daily dual touch without event order stays unknown',()=>{
 const i=fixture();i.bars[4].low='84';i.bars[4].high='121';
 const x=evaluateLaunchOutcome(i);assert.deepEqual(x.research.barrier20,{state:'order_unknown',session:i.sessions[4].session,position:5,observed:'range'});
});

test('before maturity pending and missing periods remain distinct',()=>{

});
test('prior missing bar cannot confirm a later first hit',()=>{
 const i=fixture();i.bars[10].high='180';i.bars.splice(2,1);

 assert.notEqual(a.execution.nextSession,cancelled);

});
