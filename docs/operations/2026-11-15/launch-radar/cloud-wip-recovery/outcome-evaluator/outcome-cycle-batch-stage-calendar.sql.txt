CREATE TABLE public.fixture_outcome_annual(receipt jsonb);
CREATE FUNCTION public.vnext_launch_cycle_context_v1(p_principal text) RETURNS jsonb LANGUAGE sql STABLE AS $$
 SELECT jsonb_build_object('ready',current_setting('fixture.cycle_ready',true)='yes',
  'session',substring(current_setting('fixture.cycle_cutoff',true),1,10),'cutoff',current_setting('fixture.cycle_cutoff',true))
$$;
CREATE FUNCTION public.vnext_launch_annual_rows_internal_v1(market_name text,first_day date,last_day date,cutoff timestamptz)
RETURNS jsonb LANGUAGE sql STABLE AS $$
 SELECT jsonb_agg(receipt) FROM public.fixture_outcome_annual WHERE receipt#>>'{row,market}'=market_name
  AND (receipt#>>'{row,session}')::date BETWEEN first_day AND last_day AND (receipt->>'knownAt')::timestamptz<=cutoff
$$;
