import errno
import json
import os
from pathlib import Path
import re
import stat
import subprocess
import sys
import time

def budget(parent):
    cg=Path('/proc/self/cgroup').read_text().strip()
    match=re.fullmatch(r'0::/system[.]slice/(stockinsider-stage-credential-check-[A-Za-z0-9_.-]+[.]service)',cg)
    if os.getuid()==0 or not match:
        return {'result':'UNIT_NOT_VERIFIED'}
    unit=match.group(1)
    manager_path='/run/credentials/'+unit
    if parent and os.environ.get('CREDENTIALS_DIRECTORY')!=manager_path:
        return {'result':'CREDENTIAL_DIRECTORY_MISSING'}
    directory=-1
    fd=-1
    try:
        directory=os.open(manager_path,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
        fd=os.open('close-observer-profile',os.O_RDONLY|os.O_NOFOLLOW|os.O_CLOEXEC,dir_fd=directory)
        info=os.fstat(fd)
        if not stat.S_ISREG(info.st_mode) or info.st_mode&0o222 or info.st_uid not in (0,os.geteuid()):
            return {'result':'CREDENTIAL_MODE_REJECTED'}
        data=os.read(fd,300)
        try:
            w=os.open('close-observer-profile',os.O_WRONLY|os.O_NOFOLLOW|os.O_CLOEXEC,dir_fd=directory)
        except OSError as e:
            if e.errno not in (errno.EROFS,errno.EACCES,errno.EPERM):
                return {'result':'CREDENTIAL_ERROR'}
        else:
            os.close(w)
            return {'result':'CREDENTIAL_MUTABLE'}
    except OSError:
        return {'result':'CREDENTIAL_MISSING'}
    finally:
        if fd>=0:os.close(fd)
        if directory>=0:os.close(directory)
    if data!=('v1|'+unit+'|90|134217728|12|control-group').encode():
        return {'result':'CREDENTIAL_UNIT_OR_PROFILE_MISMATCH'}
    root=Path('/sys/fs/cgroup/system.slice')/unit
    mem=int((root/'memory.max').read_text())
    pids=int((root/'pids.max').read_text())
    if (mem,pids)!=(134217728,12):
        return {'result':'KERNEL_LIMITS_MISMATCH','memoryMax':mem,'tasksMax':pids}
    if int((root/'memory.current').read_text())>134217728-(80 if parent else 16)*1024**2 or int((root/'pids.current').read_text())>(8 if parent else 10):
        return {'result':'INSUFFICIENT_HEADROOM'}
    entry=Path('/proc')/str(os.getpid() if parent else os.getppid())
    if (entry/'cgroup').read_text().strip()!=cg or entry.stat().st_uid!=os.getuid():
        return {'result':'PARENT_BOUNDARY_MISMATCH'}
    start=int((entry/'stat').read_text().rsplit(')',1)[1].split()[19])/os.sysconf('SC_CLK_TCK')
    remaining=90-(time.monotonic()-start)
    if not (30 if parent else 8)<remaining<=90:
        return {'result':'DEADLINE_REJECTED'}
    return {'result':'VERIFIED','budgetSeconds':round(remaining,3),'credentialReadOnly':True,'memoryMax':mem,'tasksMax':pids}
parent=len(sys.argv)==1
r=budget(parent)
if parent:
    child=subprocess.run([sys.executable,'-I','-B',__file__,'child'],env={'PATH':'/usr/bin:/bin','LC_ALL':'C'},capture_output=True,text=True,timeout=4)
    out={'parent':r,'child':json.loads(child.stdout) if child.returncode==0 else {'exitCode':child.returncode},'nonRoot':os.getuid()!=0}
    print(json.dumps(out),flush=True)
    time.sleep(1.2)
else:
    print(json.dumps(r),flush=True)
