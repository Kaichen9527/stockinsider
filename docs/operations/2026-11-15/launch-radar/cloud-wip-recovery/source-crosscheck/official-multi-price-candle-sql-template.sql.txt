
CREATE FUNCTION public.vnext_launch_calendar_v1(venue text, signal_day date, cutoff timestamptz) RETURNS jsonb
LANGUAGE sql STABLE AS $$
 SELECT jsonb_build_object('sessions',jsonb_agg(jsonb_build_object('close_at',(day::date+time '13:30') AT TIME ZONE 'Asia/Taipei') ORDER BY day))
 FROM generate_series(signal_day-239,signal_day,interval '1 day') day
$$;

LANGUAGE sql STABLE AS $$

 'bars',jsonb_agg(CASE WHEN i<>239 THEN jsonb_build_object('session',date '2026-11-10'+(i-239),
 'open',100,'high',100,'low',100,'close',100,'rawHigh',100,'rawLow',100)
 ELSE jsonb_build_object('session',date '2026-11-10','open',100,'high',100,'low',100,'close',100)
 || CASE WHEN current_setting('fixture.flat',true)='yes' THEN jsonb_build_object('rawHigh',100,'rawLow',100)
 ELSE __REAL_BAR__::jsonb END END ORDER BY i)) ELSE NULL END
 FROM generate_series(0,239) i
$$;
