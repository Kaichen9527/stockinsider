CREATE FUNCTION public.vnext_target_quote_v1(p_stock uuid,p_symbol text,p_exchange text,p_baseline date,p_cutoff timestamptz) RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path=pg_catalog,public SET timezone='UTC' SET statement_timeout='30s' SET lock_timeout='2s' AS $probe_body$BEGIN RETURN '{}'::jsonb; END$probe_body$;
DO $proof$ DECLARE result jsonb; BEGIN SELECT jsonb_agg(jsonb_build_object('signature',j->>'signature','config',to_jsonb(f.proconfig),'matchesOriginalProfile',to_jsonb(f.proconfig)=j->'old','matchesFixedProfile',to_jsonb(f.proconfig)=j->'repaired','securityDefiner',f.prosecdef) ORDER BY ord) INTO result FROM jsonb_array_elements('[{"signature":"public.vnext_target_current_v1(text,uuid,text)","old":["search_path=pg_catalog, public","statement_timeout=30s","lock_timeout=2s"],"repaired":["search_path=pg_catalog, public","statement_timeout=30s","lock_timeout=2s"]},{"signature":"public.vnext_read_v1(text)","old":["search_path=pg_catalog, public"],"repaired":["search_path=pg_catalog, public"]},{"signature":"public.vnext_target_quote_v1(uuid,text,text,date,timestamptz)","old":["search_path=pg_catalog, public","statement_timeout=30s","lock_timeout=2s"],"repaired":["search_path=pg_catalog, public","TimeZone=UTC","statement_timeout=30s","lock_timeout=2s"]}]'::jsonb) WITH ORDINALITY AS t(j,ord) JOIN pg_proc f ON f.oid=to_regprocedure(t.j->>'signature'); RAISE WARNING USING MESSAGE='STOCKINSIDER_PROBE_RESULT '||result::text; END $proof$;
