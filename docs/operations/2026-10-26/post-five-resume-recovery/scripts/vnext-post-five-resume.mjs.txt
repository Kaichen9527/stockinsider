const sha=b=>createHash('sha256').update(b).digest('hex');
const normal=s=>s.replace('public.','').replaceAll('timestamptz','timestamp with time zone').replaceAll(', ',',');
// Compare EVERY original value, not a digest/count. Avoid assert's enormous
// diagnostic diff of private23MiB catalog/rows on a deliberately bad snapshot.
const exact=(actual,expected,label)=>assert.ok(isDeepStrictEqual(actual,expected),label);

// Open, fstat, read and lstat the SAME regular inode. No symlink component,
// hardlink, secret path, arbitrary root or caller-provided source selector.
function pinnedRead(root,row){
 assert.match(row.path,/^[A-Za-z0-9_.\[\]/-]+$/u);assert.ok(!row.path.split('/').includes('..'));
 const full=path.resolve(root,row.path);assert.ok(full.startsWith(root+'/'));
 let p=root;for(const part of row.path.split('/')){p=path.join(p,part);assert.ok(!fs.lstatSync(p).isSymbolicLink(),'symlink refused');}
 const fd=fs.openSync(full,fs.constants.O_RDONLY|fs.constants.O_NOFOLLOW);
 try{const a=fs.fstatSync(fd,{bigint:true});assert.ok(a.isFile()&&a.nlink===1n);const bytes=fs.readFileSync(fd),b=fs.fstatSync(fd,{bigint:true}),c=fs.lstatSync(full,{bigint:true});
  for(const k of ['dev','ino','size','mtimeNs','ctimeNs']){assert.equal(b[k],a[k],'opened source changed');assert.equal(c[k],a[k],'pathname changed');}
  assert.equal(bytes.length,row.bytes,row.path);assert.equal(sha(bytes),row.sha256,row.path);return bytes;
 }finally{fs.closeSync(fd);}
}

function stableSnapshot(s){return {catalog:s.catalog,routines:s.routines,rows:s.rows,ids:s.ids,authority:s.authority};}
function unchangedRows(before,after){
 for(const [name,value]of Object.entries(before.rows))exact(after.rows[name],value,name);
 for(const [name,value]of Object.entries(after.rows))if(!Object.hasOwn(before.rows,name))assert.deepEqual(value,{rows:[],count:0,sha256:sha('[]')},name);
}
function positiveRoutines(before,after,p){
 assert.equal(after.routines.length,before.routines.length+p.newPrivateRoutines.length);
 for(const b of before.routines){const a=after.routines.find(r=>r.oid===b.oid);assert.ok(a);const modified=p.routines.find(r=>normal(r.signature)===normal(b.signature));assert.equal(a.hash,modified?.successorBodySha256??b.hash);
  const am={...a.metadata},bm={...b.metadata};delete am.prosrc;delete bm.prosrc;assert.deepEqual(am,bm,'OID/nonbody/ACL exact');}
 for(const item of p.newPrivateRoutines){const matches=after.routines.filter(r=>normal(r.signature)===normal(item.signature));assert.equal(matches.length,1);assert.equal(matches[0].hash,item.bodySha256);}
}

// Durable intent precedes a mutation. Existing intent (even known rollback)
// is never automatically sent again. Unknown is recovered by independent READ.
function createJournal(directory){
 fs.mkdirSync(directory,{mode:0o700});const st=fs.lstatSync(directory);assert.ok(st.isDirectory()&&!st.isSymbolicLink()&&st.uid===process.getuid());
 return (name,value)=>{assert.match(name,/^[A-Za-z0-9-]+$/u);const fd=fs.openSync(path.join(directory,name+'.json'),fs.constants.O_CREAT|fs.constants.O_EXCL|fs.constants.O_WRONLY,0o600);
  try{fs.writeFileSync(fd,JSON.stringify(value)+'\n');fs.fsyncSync(fd);}finally{fs.closeSync(fd);}const dir=fs.openSync(directory,fs.constants.O_RDONLY);try{fs.fsyncSync(dir);}finally{fs.closeSync(dir);}};
}

async function queryReady(client,text,values=[],status='I'){
 let onReady,onError,onEnd,timer;
 const protocol=new Promise((resolve,reject)=>{onReady=m=>resolve(m.status);onError=reject;onEnd=()=>reject(Error('owned_client_terminal_before_ready'));client.connection.once('readyForQuery',onReady);client.once('error',onError);client.once('end',onEnd);timer=setTimeout(()=>{void client.end().catch(()=>{});reject(Error('actual_ready_unknown'));},35000);});
 try{const [query,ready]=await Promise.allSettled([client.query({text,values,query_timeout:35000}),protocol]);
  if(query.status==='rejected'){Object.assign(query.reason,{observedReady:ready.status==='fulfilled'?ready.value:null,readyUncertain:ready.status==='rejected'?ready.reason.message:null});throw query.reason;}
  assert.equal(ready.status,'fulfilled','actual Ready required');assert.equal(ready.value,status,'fresh transaction state required');return query.value;
 }finally{clearTimeout(timer);client.connection.off('readyForQuery',onReady);client.off('error',onError);client.off('end',onEnd);}
}

async function executeFixed(x,d,baseline,record){
 assert.notEqual(x.writer.processID,x.observer.processID);
 // Probe FIRST. T/E caller transactions never reach BEGIN/ROLLBACK/COMMIT.
 await x.statement(x.writer,'SELECT 1 AS fresh_owned_probe');
 exact(stableSnapshot(await x.snapshot()),baseline,'exact current full predecessor before mutation');
 for(const step of d.next){
  if(x.interrupted())throw Error('owned_batch_interrupted');
  x.verifySource();const bytes=x.readStep(step);assert.equal(sha(bytes),step.sha256);const before=await x.snapshot();
  record(step.id+'-intent',{sourceSha256:step.sha256,predecessorSha256:sha(JSON.stringify(stableSnapshot(before))),noAutomaticResend:true});
  // Recheck after observer IO; a changed pathname/source cannot use a cached
  // readiness observation to dispatch a different install.
  x.verifySource();assert.deepEqual(x.readStep(step),bytes);
  try{
   if(step.id==='positive-contra-A1'){
    const result=await x.statement(x.writer,bytes.toString());assert.equal((Array.isArray(result)?result.at(-1):result).command,'COMMIT');
    const after=await x.snapshot();positiveRoutines(before,after,x.positiveProfile);unchangedRows(before,after);
   }else if(step.id==='valuation')await x.valuation();
   else {assert.equal(step.id,'complete-qualification');await x.qualification();}
   const after=await x.snapshot();unchangedRows(before,after);x.verifySource();
   record(step.id+'-committed',{ready:'I',sourceSha256:step.sha256,snapshotSha256:sha(JSON.stringify(stableSnapshot(after))),knownCommit:true});
  }catch(primary){
   record(step.id+'-primary',{code:primary.code??null,message:primary.message,observedReady:primary.observedReady??null,automaticResend:false});
   // Only an ACTUALLY observed transaction state permits owned rollback.
   if(['T','E'].includes(primary.observedReady)){try{await x.statement(x.writer,'ROLLBACK');}catch(e){primary.rollbackUncertain=e;}}
   try{record(step.id+'-reconcile-read',{snapshot:await x.snapshot(),resend:false});}catch(e){primary.readUncertain=e;}
   throw primary;
  }
 }
 return {installedIds:d.next.map(r=>r.id),caseAdmission:'READ_ONLY_OBSERVER_ONLY',current18:'NOT_RUN_REQUIRES_ORIGINAL_CASE_ISOLATION_AND_BUSINESS_GATES',profileEnabled:false};
}
